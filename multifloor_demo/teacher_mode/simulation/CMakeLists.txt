cmake_minimum_required(VERSION 3.16)
project(teacher_actuator LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(teacher_actuator SHARED teacher_actuator.cpp)
target_compile_features(teacher_actuator PRIVATE cxx_std_17)
target_compile_options(teacher_actuator PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(teacher_actuator PRIVATE gz-sim8::gz-sim8 gz-plugin2::gz-plugin2)
install(TARGETS teacher_actuator LIBRARY DESTINATION lib)
