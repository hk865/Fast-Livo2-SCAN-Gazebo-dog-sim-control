cmake_minimum_required(VERSION 3.16)
project(champ_native_observer LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(champ_native_observer SHARED champ_native_observer.cpp)
target_compile_features(champ_native_observer PRIVATE cxx_std_17)
target_compile_options(champ_native_observer PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(champ_native_observer PRIVATE gz-sim8::gz-sim8 gz-plugin2::gz-plugin2)
