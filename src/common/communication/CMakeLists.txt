cmake_minimum_required(VERSION 3.16)
project(navigation_writer_qos LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rcl REQUIRED)
find_package(rmw_fastrtps_shared_cpp REQUIRED)
find_package(fastrtps REQUIRED)
add_library(writer_qos SHARED writer_qos.cpp)
set_target_properties(writer_qos PROPERTIES PREFIX "")
ament_target_dependencies(writer_qos rcl rmw_fastrtps_shared_cpp)
target_link_libraries(writer_qos fastrtps)
install(TARGETS writer_qos LIBRARY DESTINATION lib)
